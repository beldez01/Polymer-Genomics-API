# Phase 1C/1D/1E Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add melting domain profiles and mutation dG tracks (1C), composite fragility score (1D), and a per-layer validation framework (1E) to the Polymer Genomics API.

**Architecture:** Phase 1C adds 7 new columns to the existing `biophysics.sequence_properties` table (ALTER TABLE + re-compute + re-ingest). Phase 1D adds a new `fragility.composite_score` table combining existing non-B DNA, breakpoints, biophysics curvature, and replication timing into a single score. Phase 1E creates a validation harness that runs per-layer checks (row counts, value ranges, cross-layer correlations) and emits pass/fail reports.

**Tech Stack:** PostgreSQL (asyncpg), FastAPI, Python (pure — no scipy/numpy), Poland-Scheraga model for melting, SantaLucia NN parameters for mutation dG.

**Existing state:**
- `biophysics.sequence_properties`: 2.9M rows, 15 numeric columns (7 core + 8 DNAshapeR)
- `fragility.breakpoints`: 49 rows (CFS + translocations, ingested)
- `fragility.nonb_dna`: 2.9M rows (G4, Z-DNA, cruciform, R-loop, triplex, ingested)
- No composite fragility score exists
- No validation framework exists

```sql
    ADD COLUMN IF NOT EXISTS sbs_t_to_a_ddg real;  -- T>A (complement A>T)
```

- [ ] **Step 3: Implement `compute_sbs_ddg(seq)`**

For each position in the sequence that is C (or T), compute the stacking energy change when mutated:
- Original: dG(5'XY3') + dG(5'YZ3') where Y is the mutated position
- Mutated: dG(5'XY'3') + dG(5'Y'Z3') where Y' is the new base
- delta-delta-G = |mutant_dG - wildtype_dG|
- Average across all positions of that base type in the window
- Return dict with 4 SBS channels

```sql
-- 037_fragility_composite.sql
-- Composite fragility score combining non-B DNA, biophysics, breakpoint proximity

CREATE TABLE IF NOT EXISTS fragility.composite_score (
    id bigint GENERATED ALWAYS AS IDENTITY,
    layer_id uuid NOT NULL REFERENCES registry.layers(id),
    build genome_build NOT NULL,
    chr_id smallint NOT NULL REFERENCES ref.chromosomes(chr_id),
    start_pos int NOT NULL,
    end_pos int NOT NULL,
    coord int4range GENERATED ALWAYS AS (int4range(start_pos, end_pos)) STORED,

    -- Component scores (0-1 normalized)
    nonb_component real,         -- normalized total non-B DNA density
    curvature_component real,    -- normalized curvature (high = fragile)
    stacking_component real,     -- normalized stacking instability (inverted dG)
    breakpoint_proximity real,   -- decay function from nearest known breakpoint

    -- Composite
    fragility_score real NOT NULL,  -- weighted combination [0,1]
    fragility_class text,           -- 'low', 'moderate', 'high', 'extreme'

    PRIMARY KEY (id, build, chr_id)
) PARTITION BY LIST (build);

-- hg38 partitions
CREATE TABLE IF NOT EXISTS fragility.composite_score_hg38
    PARTITION OF fragility.composite_score FOR VALUES IN ('hg38')
    PARTITION BY HASH (chr_id);

CREATE TABLE IF NOT EXISTS fragility.composite_score_hg38_p0
    PARTITION OF fragility.composite_score_hg38 FOR VALUES WITH (MODULUS 4, REMAINDER 0);
CREATE TABLE IF NOT EXISTS fragility.composite_score_hg38_p1
    PARTITION OF fragility.composite_score_hg38 FOR VALUES WITH (MODULUS 4, REMAINDER 1);
CREATE TABLE IF NOT EXISTS fragility.composite_score_hg38_p2
    PARTITION OF fragility.composite_score_hg38 FOR VALUES WITH (MODULUS 4, REMAINDER 2);
CREATE TABLE IF NOT EXISTS fragility.composite_score_hg38_p3
    PARTITION OF fragility.composite_score_hg38 FOR VALUES WITH (MODULUS 4, REMAINDER 3);

-- Indexes
CREATE INDEX IF NOT EXISTS idx_fragility_composite_range
    ON fragility.composite_score USING gist (chr_id, coord);
CREATE INDEX IF NOT EXISTS idx_fragility_composite_layer
    ON fragility.composite_score (layer_id, build);

-- Permissions
GRANT SELECT ON ALL TABLES IN SCHEMA fragility TO api_reader;
GRANT SELECT, INSERT, UPDATE ON ALL TABLES IN SCHEMA fragility TO ingest_writer;

-- Add layer_type enum if needed
DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_enum
        WHERE enumtypid = 'layer_type'::regtype AND enumlabel = 'fragility'
    ) THEN
        ALTER TYPE layer_type ADD VALUE IF NOT EXISTS 'fragility';
    END IF;
END $$;
```

```python
# Weights from roadmap: F(x) = alpha*NDR + beta*curvature + gamma*flexibility + delta*non-B-form
WEIGHTS = {
    "nonb": 0.35,        # non-B DNA density (strongest predictor per literature)
    "curvature": 0.25,   # intrinsic curvature (flexibility proxy)
    "stacking": 0.25,    # stacking instability (inverse dG)
    "bp_prox": 0.15,     # proximity to known breakpoints
}

def compute_fragility_score(nonb, curvature, stacking, bp_prox):
    return (WEIGHTS["nonb"] * nonb + WEIGHTS["curvature"] * curvature +
            WEIGHTS["stacking"] * stacking + WEIGHTS["bp_prox"] * bp_prox)
```

```python
# src/polymer_genomics/validation/framework.py
from dataclasses import dataclass

@dataclass
class ValidationResult:
    layer_key: str
    check_name: str
    passed: bool
    message: str
    actual: float | int | None = None
    expected: str | None = None

def validate_row_count(layer_key, actual, expected_min):
    passed = actual >= expected_min
    return ValidationResult(
        layer_key=layer_key,
        check_name="row_count",
        passed=passed,
        message=f"Row count {actual:,} {'≥' if passed else '<'} minimum {expected_min:,}",
        actual=actual,
        expected=f">= {expected_min:,}",
    )

def validate_value_range(field, values, expected_min, expected_max):
    out_of_range = [v for v in values if v < expected_min or v > expected_max]
    passed = len(out_of_range) == 0
    return ValidationResult(
        layer_key=field,
        check_name="value_range",
        passed=passed,
        message=f"{len(out_of_range)} values outside [{expected_min}, {expected_max}]",
        actual=len(out_of_range),
    )

def validate_null_fraction(field, null_count, total, max_null_frac):
    frac = null_count / total if total > 0 else 0
    passed = frac <= max_null_frac
    return ValidationResult(
        layer_key=field,
        check_name="null_fraction",
        passed=passed,
        message=f"Null fraction {frac:.4f} {'≤' if passed else '>'} {max_null_frac}",
        actual=frac,
    )
```

```python
async def run_layer_validation(
    conn: asyncpg.Connection,
    layer_key: str,
    build: str = "hg38",
) -> list[ValidationResult]:
    """Run all validation checks for a given layer."""
    spec = LAYER_VALIDATION_SPECS.get(layer_key)
    if spec is None:
        return [ValidationResult(layer_key, "spec_exists", False, "No validation spec defined")]

    results = []
    table = spec["table"]

    # Row count
    count = await conn.fetchval(f"SELECT count(*) FROM {table} WHERE build = $1::genome_build", build)
    results.append(validate_row_count(layer_key, count, spec["row_count_min"]))

    # Value ranges (sample 10K rows)
    for field, (vmin, vmax) in spec.get("value_ranges", {}).items():
        rows = await conn.fetch(
            f"SELECT {field} FROM {table} WHERE build = $1::genome_build AND {field} IS NOT NULL LIMIT 10000",
            build,
        )
        values = [float(r[0]) for r in rows]
        results.append(validate_value_range(f"{layer_key}.{field}", values, vmin, vmax))

    # Null fraction
    for field in spec.get("value_ranges", {}):
        null_count = await conn.fetchval(
            f"SELECT count(*) FROM {table} WHERE build = $1::genome_build AND {field} IS NULL",
            build,
        )
        results.append(validate_null_fraction(
            f"{layer_key}.{field}", null_count, count, spec.get("max_null_frac", 0.01),
        ))

    return results

async def run_all_validations(conn, build="hg38"):
    """Run validations for all layers with specs."""
    all_results = {}
    for layer_key in LAYER_VALIDATION_SPECS:
        all_results[layer_key] = await run_layer_validation(conn, layer_key, build)
    return all_results
```
